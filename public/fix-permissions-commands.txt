# Comandos para ejecutar en el servidor de producción

# 1. Cambiar al directorio del proyecto
cd /opt/bitnami/apache/htdocs/desarrollo/rinotrack/public/

# 2. Verificar el usuario actual del servidor web
ps aux | grep apache
# o
ps aux | grep httpd

# 3. Cambiar permisos del directorio uploads para que sea escribible por todos
chmod 777 uploads/

# 4. Verificar los permisos
ls -la uploads/

# 5. Si el comando anterior no funciona, intentar con sudo
sudo chmod 777 uploads/

# 6. Alternativamente, cambiar el propietario al usuario del servidor web
# (reemplaza 'www-data' por el usuario correcto del servidor web)
sudo chown www-data:www-data uploads/
sudo chmod 755 uploads/

# 7. O dar permisos de escritura al grupo
sudo chmod 775 uploads/

# 8. Verificar que funciona creando un archivo de prueba
echo "test" > uploads/test.txt
rm uploads/test.txt
